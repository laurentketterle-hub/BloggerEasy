Building a Type-Safe Event Bus in Rust

2025-03-20 • 8 min read

Event-driven architectures shine in systems programming, but type safety often takes a back seat. Let's build a compile-time checked event bus using Rust's powerful type system and trait bounds.

pub trait Event: Send + Sync + 'static {
    type Payload: Serialize + DeserializeOwned;
}

pub struct EventBus {
    handlers: HashMap<TypeId, Vec<Box<dyn Any>>>,
}

impl EventBus {
    pub fn subscribe<E: Event>(&mut self, handler: fn(&E::Payload)) {
        let id = TypeId::of::<E>();
        self.handlers
            .entry(id)
            .or_default()
            .push(Box::new(handler));
    }
}
